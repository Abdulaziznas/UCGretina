cmake_minimum_required(VERSION 3.16)
project(UCGretina LANGUAGES C CXX)

option(WITH_VIS "Enable Geant4 visualization" ON)
option(LHTARGET "Use the liquid hydrogen target" OFF)
option(SCANNING "Use the LBNL scanning table" OFF)
option(POL "Enable nuclear polarization" OFF)
option(NEUTRONS "Enable neutron and hadronic physics" OFF)
option(HIGHMULT "Enable high multiplicity output" OFF)
option(CACHETEXT "Use text cache files" OFF)

if(LHTARGET AND SCANNING)
  message(FATAL_ERROR "LHTARGET and SCANNING select different geometries; use separate builds")
endif()
if(WITH_VIS)
  find_package(Geant4 11 REQUIRED COMPONENTS ui_all vis_all)
else()
  find_package(Geant4 11 REQUIRED)
endif()
include(${Geant4_USE_FILE})

set(GIT_HASH unknown)
set(GIT_BRANCH unknown)
find_package(Git QUIET)
if(Git_FOUND AND EXISTS "${PROJECT_SOURCE_DIR}/.git")
  execute_process(COMMAND ${GIT_EXECUTABLE} rev-parse --short HEAD
    WORKING_DIRECTORY ${PROJECT_SOURCE_DIR} OUTPUT_VARIABLE GIT_HASH
    OUTPUT_STRIP_TRAILING_WHITESPACE)
  execute_process(COMMAND ${GIT_EXECUTABLE} rev-parse --abbrev-ref HEAD
    WORKING_DIRECTORY ${PROJECT_SOURCE_DIR} OUTPUT_VARIABLE GIT_BRANCH
    OUTPUT_STRIP_TRAILING_WHITESPACE)
endif()
configure_file(cmake/Git_Hash.hh.in generated/Git_Hash.hh @ONLY)

set(executable UCGretina)
if(LHTARGET)
  string(APPEND executable _LH)
elseif(SCANNING)
  string(APPEND executable _Scan)
endif()
if(POL)
  string(APPEND executable _Pol)
endif()

file(GLOB sources CONFIGURE_DEPENDS src/*.cc)
add_executable(${executable} ${executable}.cc ${sources})
target_compile_features(${executable} PRIVATE cxx_std_17)
target_include_directories(${executable} PRIVATE
  ${PROJECT_BINARY_DIR}/generated ${PROJECT_SOURCE_DIR}/include)
# G4V10 selects the geometry implementation used by Geant4 10 and later.
target_compile_definitions(${executable} PRIVATE G4V10 _FILE_OFFSET_BITS=64)
foreach(feature LHTARGET SCANNING POL NEUTRONS HIGHMULT CACHETEXT)
  if(${feature})
    target_compile_definitions(${executable} PRIVATE ${feature})
  endif()
endforeach()
if(WITH_VIS)
  target_compile_definitions(${executable} PRIVATE G4VIS_USE)
endif()
target_link_libraries(${executable} PRIVATE ${Geant4_LIBRARIES})
install(TARGETS ${executable} RUNTIME DESTINATION bin)

include(CTest)
if(BUILD_TESTING)
  find_package(Python3 REQUIRED COMPONENTS Interpreter)
  set(smoke_cases)
  if(SCANNING)
    list(APPEND smoke_cases scan)
  elseif(LHTARGET)
    list(APPEND smoke_cases lh)
  else()
    list(APPEND smoke_cases standard sources background atima)
    if(POL)
      list(APPEND smoke_cases pol)
    endif()
  endif()
  if(NEUTRONS)
    list(APPEND smoke_cases neutron)
  endif()
  set(thread_args)
  if(Geant4_multithreaded_FOUND)
    set(thread_args --threads 2)
  endif()
  foreach(smoke_case IN LISTS smoke_cases)
    add_test(NAME smoke_${smoke_case}
      COMMAND ${Python3_EXECUTABLE} ${PROJECT_SOURCE_DIR}/tests/geant4_smoke.py
        --binary $<TARGET_FILE:${executable}> --case ${smoke_case} ${thread_args})
    set_tests_properties(smoke_${smoke_case} PROPERTIES TIMEOUT 180)
  endforeach()
endif()
